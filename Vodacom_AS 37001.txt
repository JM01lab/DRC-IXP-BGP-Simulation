! Vodacom ISP - Edge Router Configuration
!
hostname Vodacom
!
interface GigabitEthernet0/0
 description Link to IXP_DRC
 ip address 15.0.0.1 255.255.255.0
!
interface GigabitEthernet0/1
 description Local Area Network (LAN) for Clients
 ip address 192.168.1.1 255.255.255.0
!
! BGP Routing Logic
router bgp 37001
 bgp log-neighbor-changes
 ! Peering with the IXP Hub
 neighbor 15.0.0.2 remote-as 65000
 !
 address-family ipv4
  neighbor 15.0.0.2 activate
  ! Important: Ensure the router is the gateway for its internal routes
  neighbor 15.0.0.2 next-hop-self
  ! Announcing the customer network to the BGP table
  network 192.168.1.0 mask 255.255.255.0
 exit-address-family
!
! Note: The default route (0.0.0.0) is received dynamically 
! from the IXP via BGP redistribution.
!
